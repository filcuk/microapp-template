      Syntax highlighting via Prism. Use the controls below to switch interaction mode and toolbar position. Toolbar actions and hover copy/maximise are configured with data attributes.

      def fibonacci(n: int) -> list[int]:
    """Return the Fibonacci sequence up to n terms."""
    if n <= 0:
        return []
    if n == 1:
        return [0]

    sequence = [0, 1]
    while len(sequence) < n:
        sequence.append(sequence[-1] + sequence[-2])
    return sequence


def main() -> None:
    terms = fibonacci(12)
    print("Fibonacci sequence:")
    for index, value in enumerate(terms):
        print(f"  F({index}) = {value}")


if __name__ == "__main__":
    main()
